================================================================================
                    将本地服务部署到 K8S 集群完整步骤
================================================================================

步骤概览:
本地服务 → Docker 镜像 → 传输到集群 → 导入 containerd → K8s 部署

================================================================================
步骤 1: 创建 Web API 包装器（如果需要）
================================================================================

如果本地服务是脚本而非 Web 服务，需要创建 FastAPI 包装器：

# api.py
from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI(title="服务名称")

@app.get("/health")
async def health():
    return {"status": "healthy", "service": "服务名称"}

# 添加业务 API 端点...

================================================================================
步骤 2: 创建 requirements.txt
================================================================================

fastapi>=0.100.0
uvicorn>=0.23.0
pydantic>=2.0.0
# 其他依赖...

================================================================================
步骤 3: 创建 Dockerfile
================================================================================

FROM python:3.10-slim

WORKDIR /app

# 安装系统依赖（如需要）
RUN apt-get update && apt-get install -y --no-install-recommends \\
    build-essential \\
    && rm -rf /var/lib/apt/lists/*

# 安装 Python 依赖
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# 复制代码
COPY . .

EXPOSE 8000

CMD ["uvicorn", "api:app", "--host", "0.0.0.0", "--port", "8000"]

================================================================================
步骤 4: 创建 K8s 部署文件
================================================================================

# deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: 服务名称
  namespace: aiops
spec:
  replicas: 1
  selector:
    matchLabels:
      app: 服务名称
  template:
    spec:
      nodeName: k8s-master  # 指定节点
      containers:
      - name: 服务名称
        image: aiops/服务名称:latest
        imagePullPolicy: IfNotPresent
        ports:
        - containerPort: 8000
        resources:
          requests:
            cpu: "500m"
            memory: "1Gi"
          limits:
            cpu: "2000m"
            memory: "4Gi"
---
apiVersion: v1
kind: Service
metadata:
  name: 服务名称-nodeport
  namespace: aiops
spec:
  type: NodePort
  selector:
    app: 服务名称
  ports:
  - port: 8000
    targetPort: 8000
    nodePort: 30083  # 选择未使用的端口

================================================================================
步骤 5: 构建 Docker 镜像
================================================================================

# 构建 amd64 架构镜像（集群节点是 amd64）
cd /path/to/service
docker buildx build --platform linux/amd64 -t aiops/服务名称:latest --load .

================================================================================
步骤 6: 导出镜像
================================================================================

# 导出为 tar 文件
docker save aiops/服务名称:latest -o /tmp/服务名称.tar

================================================================================
步骤 7: 传输到集群
================================================================================

# 传输镜像和部署文件
scp /tmp/服务名称.tar jaci@8.136.226.231:/tmp/
scp deployment.yaml jaci@8.136.226.231:/tmp/

================================================================================
步骤 8: 导入镜像到 containerd
================================================================================

# SSH 到集群节点
ssh jaci@8.136.226.231

# 导入镜像到 k8s.io 命名空间
sudo ctr -n k8s.io images import /tmp/服务名称.tar

================================================================================
步骤 9: 部署到 K8s
================================================================================

# 应用部署文件
sudo kubectl apply -f /tmp/deployment.yaml

# 检查 Pod 状态
sudo kubectl get pods -n aiops -l app=服务名称

================================================================================
步骤 10: 测试服务
================================================================================

# 健康检查
curl http://8.136.226.231:30083/health

open http://8.136.226.231:30083/docs

常用调试命令
================================================================================

# 查看 Pod 日志
sudo kubectl logs -n aiops -l app=服务名称 --tail=100

# 查看 Pod 详情

# 进入 Pod 调试

# 查看服务状态
sudo kubectl get svc -n aiops

================================================================================
一键部署脚本示例
================================================================================

#!/bin/bash
SERVICE_NAME="my-service"
SERVICE_PORT=30083
SERVICE_DIR="/path/to/service"

# 构建
docker buildx build --platform linux/amd64 -t aiops/${SERVICE_NAME}:latest --load ${SERVICE_DIR}

# 导出
docker save aiops/${SERVICE_NAME}:latest -o /tmp/${SERVICE_NAME}.tar

# 传输
scp /tmp/${SERVICE_NAME}.tar jaci@8.136.226.231:/tmp/
scp ${SERVICE_DIR}/deployment.yaml jaci@8.136.226.231:/tmp/

# 部署
ssh jaci@8.136.226.231 "sudo ctr -n k8s.io images import /tmp/${SERVICE_NAME}.tar && sudo kubectl apply -f /tmp/deployment.yaml"
